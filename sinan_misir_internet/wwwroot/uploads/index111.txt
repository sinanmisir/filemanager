$username = "Kullanıcı";

$gender = "Male";
$user_image = "ui/images/user_male.jpg";

    $query = "SELECT username, status, gender, image FROM users WHERE userid = :userid LIMIT 1";

async function startScreenSharing() {
    try {
        const peerId = getCurrentPeerId();
        screenStream = await navigator.mediaDevices.getDisplayMedia({
            video: { displaySurface: 'monitor', frameRate: 30 },
            audio: true
        });

        // Yeni video track'i ekleyin
        const screenTrack = screenStream.getVideoTracks()[0];
        const sender = peerConnection.getSenders().find(s => 
            s.track && s.track.kind === 'video'
        );
        
        if (sender) {
            await sender.replaceTrack(screenTrack);
        } else {
            // Ekran paylaşımı için yeni transceiver ekleyin
            peerConnection.addTransceiver(screenTrack, {
                direction: 'sendonly',
                streams: [screenStream]
            });
        }

        // Yeni offer oluştur
        const offer = await peerConnection.createOffer({
            offerToReceiveVideo: true // Video almayı zorunlu kıl
        });
        await peerConnection.setLocalDescription(offer);
        
        socket.emit('screenShareOffer', {
            to: peerId,
            offer: offer,
            isScreen: true // Ekran paylaşımı olduğunu belirt
        });

        // UI güncelleme
        showScreenPreview(screenStream);
        socket.emit('screenShareStarted', peerId);

    } catch (err) {
        console.error('Ekran paylaşım hatası:', err);
        stopScreenSharing();
    }
}

function stopScreenSharing() {
    if (screenStream) {
        screenStream.getTracks().forEach(track => track.stop());
        socket.emit('screenShareStopped', getCurrentPeerId());
        
        // Kamera görüntüsüne geri dön
        const videoSender = peerConnection.getSenders()
            .find(s => s.track?.kind === 'video');
        if (videoSender && localStream) {
            const videoTrack = localStream.getVideoTracks()[0];
            videoSender.replaceTrack(videoTrack);
        }
    }
    hideScreenContainer();
}

function updateCallTimer() {
            const now = new Date();
            const diff = now - startTime;
            const hours = Math.floor(diff / 3600000);
            const minutes = Math.floor((diff % 3600000) / 60000);
            const seconds = Math.floor((diff % 60000) / 1000);
            
            document.getElementById('callTimer').textContent = 
                `${hours.toString().padStart(2, '0')}:${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
}

    function handle_result(result, type) {
        if(result.trim() != "") {
            var inner_content = _("main_content");
            var obj = JSON.parse(result);
            if(obj.message_type == "info") {
                alert(obj.message);
                return;
            }

            switch(obj.data_type) {
                case "user_info":
                    var user_info = _("user_info_content");
                    user_info.innerHTML = obj.message;
                    break;

                case "contacts":
                case "messages":
                case "groups":
                case "settings":
                case "friends":
                    inner_content.innerHTML = obj.message;
                    break;

                case "send_message":
                    var messages_holder = _("messages_holder");
                    messages_holder.innerHTML = obj.message;
                    break;
            }
        }
    }

function updateMessageNotifications(messages) {
    let hasNewMessages = false;
    let contacts = document.querySelectorAll('.contact');
    
    contacts.forEach(contact => {
        let userid = contact.getAttribute('onclick').match(/openChat\("([^"]+)"/)[1];
        let messageData = messages.find(m => m.userid === userid);
        
        if(messageData && messageData.unread_count > 0) {
            hasNewMessages = true;
            let nameDiv = contact.querySelector('.contact_name');
            let existingBadge = nameDiv.querySelector('.unread_badge');
            
            if(existingBadge) {
                existingBadge.textContent = messageData.unread_count;
            } else {
                let badge = document.createElement('span');
                badge.className = 'unread_badge';
                badge.textContent = messageData.unread_count;
                nameDiv.appendChild(badge);
            }
            
            // Update last message
            let lastMessageDiv = contact.querySelector('.last_message');
            if(lastMessageDiv && messageData.last_message) {
                lastMessageDiv.textContent = messageData.last_message;
            }
        }
    });if(hasNewMessages) {
        let sound = document.getElementById('notification_sound');
        sound.play().catch(e => console.log('Error playing sound:', e));
    }

}

async function acceptCall(data) {
  try {
    stopAllCallSounds();
    // Kullanıcının ses akışını al
    localStream = await navigator.mediaDevices.getUserMedia({ audio: true });
    
    // Yeni RTCPeerConnection oluştur
    peerConnection = new RTCPeerConnection(configuration);
    
    // Yerel akışı ekle
    localStream.getTracks().forEach(track => {
      peerConnection.addTrack(track, localStream);
    });
    
    // Local audio elementini güncelle
    const localAudio = document.getElementById('local_audio');
    localAudio.srcObject = localStream;
    
    // Remote stream geldiğinde remote audio elementini güncelle
    peerConnection.ontrack = event => {
      const remoteAudio = document.getElementById('remote_audio');
      if (!remoteAudio.srcObject) {
        remoteAudio.srcObject = event.streams[0];
      }
    };
    
    // ICE adaylarını karşı tarafa gönder
    peerConnection.onicecandidate = event => {
      if (event.candidate) {
        socket.emit('iceCandidate', {
          to: data.from,
          candidate: event.candidate
        });
      }
    };
    
    // Gelen offer'ı remote description olarak ayarla
    // Bazı tarayıcılarda new RTCSessionDescription ile sarmak gerekebilir:
    await peerConnection.setRemoteDescription(new RTCSessionDescription(data.offer));
    
    // Buffer'da bekleyen ICE adaylarını ekle
    for (const candidate of pendingIceCandidates) {
      try {
        await peerConnection.addIceCandidate(new RTCIceCandidate(candidate));
      } catch (e) {
        console.error("Buffered ICE adayı eklenirken hata:", e);
      }
    }
    // Buffer'ı temizle
    pendingIceCandidates = [];
    
    // Answer oluştur ve local description olarak ayarla
    const answer = await peerConnection.createAnswer();
    await peerConnection.setLocalDescription(answer);
    
    // Oluşturulan answer'ı karşı tarafa gönder
    socket.emit('answerCall', {
      to: data.from,
      answer: answer
    });
    
    // Görüşme sırasında görüntülenecek UI'yı göster
    showCallStatus();
    showInCallUI();
    
  } catch (error) {
    stopAllCallSounds();
    console.error('Çağrı kabul edilirken hata:', error);
    endCall();
  }
}

    try {

socket.on('iceCandidate', async (data) => {
  try {
    // Eğer peerConnection henüz oluşturulmadıysa adayları buffer'a ekle
    if (!peerConnection) {
      pendingIceCandidates.push(data.candidate);
    } else {
      // PeerConnection hazırsa doğrudan ICE adayını ekle
      await peerConnection.addIceCandidate(new RTCIceCandidate(data.candidate));
    }
  } catch (error) {
    console.error('ICE adayı eklenirken hata:', error);
  }
});

        if (!remoteScreen.srcObject) {
            remoteScreen.srcObject = event.streams[0];
        }
         }
         
         if (event.track.kind === 'audio') {

            remoteAudio.srcObject = event.streams[0];
            remoteAudio.autoplay = true;
            document.body.appendChild(remoteAudio);
