groupVoiceConnection.on("cameraShareSignal", async (data) => {
    if (!data.signal.isCameraShare) return;

    const { from, signal } = data;
    if (from === myUserId) return;

    let pc = cameraSharingPeers[from];
    if (!pc) {
        pc = new RTCPeerConnection({
            iceServers: [{ urls: "stun:stun.l.google.com:19302" }]
        });
        cameraSharingPeers[from] = pc;

        pc.pendingCandidates = [];

        pc.ontrack = (event) => {
            const userCard = document.getElementById("user_card_" + from);
            if (!userCard) return;

            const existing = userCard.querySelector("#cameraVideo_" + from);
            if (existing) existing.remove();

            const video = document.createElement("video");
            video.id = "cameraVideo_" + from;
            video.autoplay = true;
            video.playsInline = true;
            video.controls = true;
            video.style.width = "100%";
            video.style.marginTop = "10px";
            video.style.borderRadius = "12px";
            video.style.boxShadow = "0 0 10px rgba(0,0,0,0.4)";

            const stream = (event.streams && event.streams[0]) ? event.streams[0] : new MediaStream();
            video.srcObject = stream;

            userCard.appendChild(video);

            video.play().catch(err => console.error("Video oynatma hatası:", err));
        };

        pc.onicecandidate = (event) => {
            if (event.candidate) {
                groupVoiceConnection.emit("cameraShareSignal", {
                    groupId: currentGroupId,
                    from: myUserId,
                    to: from,
                    signal: {
                        type: "candidate",
                        candidate: event.candidate,
                        isCameraShare: true
                    }
                });
            }
        };
    }

    try {
        if (signal.type === "offer") {

            if (!userStatuses[from]) userStatuses[from] = {};
            userStatuses[from].cameraSharing = true;

            const userCard = document.getElementById("user_card_" + from);
            if (userCard) applyUserStatusToCard(from, userCard);
            await pc.setRemoteDescription(new RTCSessionDescription(signal));
            while (pc.pendingCandidates.length) {
                const candidate = pc.pendingCandidates.shift();
                await pc.addIceCandidate(candidate);
            }

            const answer = await pc.createAnswer();
            await pc.setLocalDescription(answer);

            groupVoiceConnection.emit("cameraShareSignal", {
                groupId: currentGroupId,
                from: myUserId,
                to: from,
                signal: {
                    type: pc.localDescription.type,
                    sdp: pc.localDescription.sdp,
                    isCameraShare: true
                }
            });

        } else if (signal.type === "answer") {
            await pc.setRemoteDescription(new RTCSessionDescription(signal));
            while (pc.pendingCandidates.length) {
                const candidate = pc.pendingCandidates.shift();
                await pc.addIceCandidate(candidate);
            }
        } else if (signal.type === "candidate" && signal.candidate) {
            if (!pc.remoteDescription || !pc.remoteDescription.type) {
                pc.pendingCandidates.push(signal.candidate);
            } else {
                await pc.addIceCandidate(signal.candidate);
            }
        }
    } catch (err) {
        console.error("Error processing camera share signal:", err);
    }
});

window.createPeerConnection = function(remoteUserId, isInitiator) {
    if (peerConnections[remoteUserId]) return peerConnections[remoteUserId];

    const pc = new RTCPeerConnection({
        iceServers: [
            { urls: "stun:stun.l.google.com:19302" },
            { urls: "stun:stun1.l.google.com:19302" },
            { urls: "stun:stun2.l.google.com:19302" },
            { urls: "stun:stun3.l.google.com:19302" },
            { urls: "stun:stun4.l.google.com:19302" }
        ]
    });
    pc.isNegotiating = false;
    pc.pendingCandidates = [];
    
    // Yerel akışın (ses) tüm parçalarını ekliyoruz.
    localStream.getTracks().forEach(track => pc.addTrack(track, localStream));
    
    pc.onicecandidate = event => {
        if (event.candidate) {
            groupVoiceConnection.emit("groupVoiceSignal", {
                groupId: currentGroupId,
                userId: myUserId,
                target: remoteUserId,
                signal: {
                    type: "candidate",
                    candidate: event.candidate
                }
            });
        }
    };

    pc.ontrack = event => {
        let audioEl = document.getElementById("remoteAudio_" + remoteUserId);
        if (!audioEl) {
            audioEl = document.createElement("audio");
            audioEl.id = "remoteAudio_" + remoteUserId;
            audioEl.autoplay = true;
            document.body.appendChild(audioEl);
        }
        let stream = (event.streams && event.streams[0]) ? event.streams[0] : new MediaStream();
        if (!event.streams || event.streams.length === 0) {
            stream.addTrack(event.track);
        }
        audioEl.srcObject = stream;

        if (event.track.kind === "audio") {
            const audioContext = new AudioContext();
            const analyser = audioContext.createAnalyser();
            const source = audioContext.createMediaStreamSource(stream);
            source.connect(analyser);
            analyser.fftSize = 256;

            const bufferLength = analyser.frequencyBinCount;
            const dataArray = new Uint8Array(bufferLength);

            function checkVolume() {
                analyser.getByteFrequencyData(dataArray);
                const volume = dataArray.reduce((a, b) => a + b) / bufferLength;

                const userCard = document.getElementById("user_card_" + remoteUserId);
                if (userCard) {
                    if (volume > 10) { // Ses var
                        userCard.style.boxShadow = "0 0 0 3px #3ba55d";
                    } else { // Sessiz
                        userCard.style.boxShadow = "none";
                    }
                }

                requestAnimationFrame(checkVolume);
            }

            checkVolume();
        }
        audioEl.play().catch(err => console.error("Audio play error:", err));
    };

    peerConnections[remoteUserId] = pc;
    
    if (isInitiator) {
        pc.isNegotiating = true;
        pc.createOffer()
          .then(offer => pc.setLocalDescription(offer))
          .then(() => {
              groupVoiceConnection.emit("groupVoiceSignal", {
                  groupId: currentGroupId,
                  userId: myUserId,
                  target: remoteUserId,
                  signal: pc.localDescription
              });
          })
          .catch(err => console.error("Offer creation error:", err))
          .finally(() => {
              pc.isNegotiating = false;
          });
    }
    
    return pc;
};

window.waitForElement = function waitForElement(selector, callback, timeout = 3000) {
    const start = Date.now();
    const interval = setInterval(() => {
        const el = document.querySelector(selector);
        if (el) {
            clearInterval(interval);
            callback(el);
        } else if (Date.now() - start > timeout) {
            clearInterval(interval);
            console.warn("Element bulunamadı (timeout):", selector);
        }
    }, 100);
}

server.js:


const express = require('express');
const https = require('https');
const fs = require('fs');
const { Server } = require('socket.io');

const app = express();

// SSL sertifika ayarları
const options = {
    key: fs.readFileSync('C:/ProgramData/win-acme/acme-v02.api.letsencrypt.org/Certificates/epikey.key'),
    cert: fs.readFileSync('C:/ProgramData/win-acme/acme-v02.api.letsencrypt.org/Certificates/episertifika.crt'),
    ca: fs.readFileSync('C:/ProgramData/win-acme/acme-v02.api.letsencrypt.org/Certificates/epicordpem.pem'),
    minVersion: 'TLSv1.2',
    ciphers: 'ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384',
    honorCipherOrder: true,
    requestCert: false,
    rejectUnauthorized: false
};

const server = https.createServer(options, app);

const io = new Server(server, {
    cors: {
        origin: "https://www.epinetwork.shop",
        methods: ["GET", "POST"],
        allowedHeaders: ["*"],
        credentials: true
    },
    allowEIO3: true,
    transports: ['websocket', 'polling'],
    path: '/socket.io/',
    pingTimeout: 60000,
    pingInterval: 25000,
    cookie: false
});

const users = new Map();
const calls = new Map();
const groupVoiceRooms = new Map();
const screenSharing = new Map();
const micMutedStatus = new Map();
const cameraSharing = new Map();

app.use(express.static('public'));

server.on('error', (err) => {
    console.error('Server error:', err);
});

io.engine.on("connection_error", (err) => {
    console.log("Connection error:", err.req);
    console.log("Connection error code:", err.code);
    console.log("Connection error message:", err.message);
    console.log("Connection error context:", err.context);
});

io.on('connection', (socket) => {
    console.log('User connected:', socket.id);
    
    socket.on('register', (userId) => {
        users.set(userId, socket.id);
        console.log('User registered:', userId);
    });

    // Grup sesli sohbet olayları
    socket.on('joinGroupVoice', (data) => {
        const { groupId, userId } = data;
        socket.join(`group-voice-${groupId}`);
        
        if (!groupVoiceRooms.has(groupId)) {
            groupVoiceRooms.set(groupId, new Set());
        }
        groupVoiceRooms.get(groupId).add(userId);
        
        socket.to(`group-voice-${groupId}`).emit('userJoinedVoice', {
            userId,
            participants: Array.from(groupVoiceRooms.get(groupId))
        });
        
        socket.emit('currentVoiceParticipants', {
            participants: Array.from(groupVoiceRooms.get(groupId)),
            screenSharing: screenSharing.get(groupId),
            micMutedUsers: micMutedStatus.has(groupId) ? Array.from(micMutedStatus.get(groupId)) : [],
            cameraSharing: cameraSharing.get(groupId)
        });
    });

    socket.on('leaveGroupVoice', (data) => {
        const { groupId, userId } = data;
        
        socket.leave(`group-voice-${groupId}`);
        
        if (groupVoiceRooms.has(groupId)) {
            groupVoiceRooms.get(groupId).delete(userId);
            
            if (screenSharing.get(groupId)?.userId === userId) {
                screenSharing.delete(groupId);
                io.to(`group-voice-${groupId}`).emit('screenSharingEnded', { userId });
            }
            
            io.to(`group-voice-${groupId}`).emit('userLeftVoice', {
                userId,
                participants: Array.from(groupVoiceRooms.get(groupId))
            });
            
            if (groupVoiceRooms.get(groupId).size === 0) {
                groupVoiceRooms.delete(groupId);
                screenSharing.delete(groupId);
            }
        }
    });

    socket.on('sendReaction', (data) => {
        io.to(`group-voice-${data.groupId}`).emit('receiveReaction', {
            userId: data.userId,
            emoji: data.emoji
        });
    });

    socket.on('cameraShareSignal', (data) => {
        const { groupId, from, to, signal } = data;
        const targetSocketId = users.get(to);
        if (targetSocketId) {
            io.to(targetSocketId).emit('cameraShareSignal', {
                groupId,
                from,
                signal: {
                    ...signal,
                    isCameraShare: true
                }
            });
        }
    });
    
    socket.on('stopCameraShare', (data) => {
        const { groupId, userId } = data;
        io.to(`group-voice-${groupId}`).emit('cameraShareEnded', { userId });
    });

    // Güncellenmiş ekran paylaşımı olayları
    socket.on('startScreenShare', (data) => {
        const { groupId, userId } = data;
        
        screenSharing.set(groupId, {
            userId,
            socketId: socket.id
        });
        
        // Odadaki tüm kullanıcılara ekran paylaşımının başladığını bildir
        io.to(`group-voice-${groupId}`).emit('screenShareStarted', {
            userId,
            signal: data.signal
        });
    });

    socket.on('screenShareSignal', (data) => {
        const { groupId, from, to, signal } = data;
        console.log("[Server] screenShareSignal received:", {
            groupId,
            from,
            to,
            signalType: signal.type,
            isScreenShare: signal.isScreenShare
        });

        const targetSocketId = users.get(to);
        if (targetSocketId) {
            // Sinyali hedef kullanıcıya ilet ve tüm gerekli bilgileri ekle
            io.to(targetSocketId).emit('screenShareSignal', {
                groupId,
                from,
                signal: {
                    ...signal,
                    isScreenShare: true
                }
            });
            console.log("[Server] Emitted screenShareSignal to:", targetSocketId);
        } else {
            console.log("[Server] Target user not found for ID:", to);
        }

        if (signal.type === 'offer' && signal.isScreenShare) {
            io.to(`group-voice-${groupId}`).emit('updateUserStatus', {
                userId: from,
                screenSharing: true
            });
        }
    });

    socket.on('screenShareAnswer', (data) => {
        const { groupId, from, to, signal } = data;
        const targetSocketId = users.get(to);
        
        if (targetSocketId) {
            io.to(targetSocketId).emit('screenShareSignal', {
                groupId,
                from,
                signal: {
                    ...signal,
                    isScreenShare: true,
                    type: 'answer'
                }
            });
        }
    });

    socket.on('stopScreenShare', (data) => {
        const { groupId, userId } = data;
        screenSharing.delete(groupId);
        io.to(`group-voice-${groupId}`).emit('screenShareEnded', { userId });

        io.to(`group-voice-${groupId}`).emit('updateUserStatus', {
            userId: userId,
            screenSharing: false
        });
    });

    socket.on('groupVoiceSignal', (data) => {
        const { groupId, userId, signal, target } = data;
        socket.to(`group-voice-${groupId}`).emit('groupVoiceSignal', {
            userId,
            target,
            signal
        });
    });

    socket.on('send_message', async (data) => {
        try {
            const response = await fetch('https://www.epinetwork.shop/api.php', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'Cookie': document.cookie
                },
                body: JSON.stringify({
                    data_type: "send_message",
                    find: {
                        userid: data.receiver,
                        message: data.message,
                        replied_to: data.replied_to
                    }
                })
            });
            
            const result = await response.json();
            
            if(result.data_type === "send_message") {
                const receiverSocket = users.get(data.receiver);
                const senderSocket = users.get(data.sender);
                
                const messageData = {
                    id: result.message_id,
                    sender: data.sender,
                    receiver: data.receiver,
                    message: data.message,
                    time: new Date().toLocaleTimeString('tr-TR', { hour: '2-digit', minute: '2-digit' }),
                    sender_name: result.sender_username,
                    sender_image: result.sender_image,
                    replied_message: result.replied_message,
                    replied_username: result.replied_username
                };
                
                if(receiverSocket) io.to(receiverSocket).emit('new_message', messageData);
                if(senderSocket) io.to(senderSocket).emit('new_message', messageData);
            }
        } catch (error) {
            console.error('Mesaj gönderme hatası:', error);
        }
    });

    socket.on("micToggle", ({ groupId, userId, muted }) => {
        io.to(`group-voice-${groupId}`).emit("micToggle", {
            userId,
            muted
        });
    
        io.to(`group-voice-${groupId}`).emit("updateUserStatus", {
            userId,
            micMuted: muted
        });
    
        if (!micMutedStatus.has(groupId)) {
            micMutedStatus.set(groupId, new Set());
        }
        const groupMuted = micMutedStatus.get(groupId);
        if (muted) {
            groupMuted.add(userId);
        } else {
            groupMuted.delete(userId);
        }
    });

    socket.on('call', (data) => {
        const { from, to } = data;
        const toSocketId = users.get(to);
        
        if (toSocketId) {
            io.to(toSocketId).emit('incomingCall', {
                from,
                offer: data.offer
            });
            calls.set(from, to);
        }
    });

    socket.on('answerCall', (data) => {
        const { to, answer } = data;
        const toSocketId = users.get(to);
        
        if (toSocketId) {
            io.to(toSocketId).emit('callAnswered', {
                answer
            });
        }
    });

    socket.on('iceCandidate', (data) => {
        const { to, candidate } = data;
        const toSocketId = users.get(to);
        
        if (toSocketId) {
            io.to(toSocketId).emit('iceCandidate', {
                candidate
            });
        }
    });

    socket.on('endCall', (data) => {
        const { to } = data;
        const toSocketId = users.get(to);
        
        if (toSocketId) {
            io.to(toSocketId).emit('callEnded');
        }
        calls.delete(data.from);
    });

    socket.on('disconnect', () => {
        let userId;
        for (const [key, value] of users.entries()) {
            if (value === socket.id) {
                userId = key;
                break;
            }
        }
        
        if (userId) {
            users.delete(userId);
            
            for (const [groupId, participants] of groupVoiceRooms.entries()) {
                if (participants.has(userId)) {
                    participants.delete(userId);
                    
                    if (screenSharing.get(groupId)?.userId === userId) {
                        screenSharing.delete(groupId);
                        io.to(`group-voice-${groupId}`).emit('screenSharingEnded', { userId });
                    }
                    
                    io.to(`group-voice-${groupId}`).emit('userLeftVoice', {
                        userId,
                        participants: Array.from(participants)
                    });
                    
                    if (participants.size === 0) {
                        groupVoiceRooms.delete(groupId);
                        screenSharing.delete(groupId);
                    }
                }
            }
            
            for (const [caller, receiver] of calls.entries()) {
                if (caller === userId || receiver === userId) {
                    const otherUser = caller === userId ? receiver : caller;
                    const otherSocketId = users.get(otherUser);
                    if (otherSocketId) {
                        io.to(otherSocketId).emit('callEnded');
                    }
                    calls.delete(caller);
                }
            }
        }
        console.log('User disconnected:', socket.id);
    });
});

const PORT = 8443;
server.listen(PORT, '0.0.0.0', () => {
    console.log(`Server running on port ${PORT}`);
